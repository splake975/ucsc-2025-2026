  \q{20}{Resolution}
  \begin{qparts}
  \item
    Which of the following clauses can be resolved with each other, and what are the resulting clauses?
    \begin{enumerate}[(A)]
      \item $x \lor y \lor z$
      \item $y$
      \item $\lnot y$
      \item $\lnot y \lor z$
    \end{enumerate}

    \begin{answer}
      \begin{enumerate}
        \item clauses A,C, results in resolution of \(x\lor z\)
        \item clauses A,D results in \(x\lor z\)
        \item clauses B,C results in \(\bot\)
        \item clauses B,D results in \(z\)
      \end{enumerate}
    \end{answer}

  \item
    Give a resolution refutation for the UNSAT formula with the following clauses:
    \begin{align*}
      c_1=\;&\lnot x \lor \lnot y \lor z \\
      c_2=\;&\lnot x \lor w \\
      c_3=\;&x \lor \lnot y \\
      c_4=\;&\lnot y \lor \lnot z \lor \lnot w \\
      c_5=\;&x \lor y \\
      c_6=\;&y \lor \lnot w
    \end{align*}
    You may write  $c_i \odot c_j$ to denote the resolution of two clauses.

    \begin{answer}
      \begin{enumerate}[i.]
        \item $c_7 = c_3 \odot c_5 = (x \lor \lnot y) \odot (x \lor y) = x$

        \item $c_8 = c_2 \odot c_7 = (\lnot x \lor w) \odot (x) = w$

        \item $c_9 = c_6 \odot c_8 = (y \lor \lnot w) \odot (w) = y$

        \item $c_{10} = c_1 \odot c_7 = (\lnot x \lor \lnot y \lor z) \odot (x) = \lnot y \lor z$

        \item $c_{11} = c_4 \odot c_9 = (\lnot y \lor \lnot z \lor \lnot w) \odot (y) = \lnot z \lor \lnot w$

        \item $c_{12} = c_{11} \odot c_8 = (\lnot z \lor \lnot w) \odot (w) = \lnot z$

        \item $c_{13} = c_1 \odot c_{12} = (\lnot x \lor \lnot y \lor z) \odot (\lnot z) = \lnot x \lor \lnot y$

        \item $c_{14} = c_{13} \odot c_7 = (\lnot x \lor \lnot y) \odot (x) = \lnot y$

        \item $c_{15} = c_{14} \odot c_9 = (\lnot y) \odot (y) = \bot$
      \end{enumerate}

    \end{answer}
  \end{qparts}
