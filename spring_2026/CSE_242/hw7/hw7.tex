    \item

    \begin{equation}
        \frac{\partial \mathcal{L}}{\partial \brac{z}{\ell}} = \frac{\partial \mathcal{L}}{\partial \brac{z}{L}} \prod_{k=\ell+1}^{L} \frac{\partial \brac{z}{k}}{\partial \brac{z}{k-1}}
    \end{equation}
    Since \(\brac{z}{k} = \brac{W}{k}\brac{a}{k-1} + \brac{b}{k}\) and \(\brac{a}{k-1} = \sigma(\brac{z}{k-1}),\) we have \(\frac{\partial \brac{z}{k}}{\partial \brac{z}{k-1}} = \brac{W}{k} \sigma'(\brac{z}{k-1})\). Substituting this back gives:
    
    \begin{equation}
        \frac{\partial \mathcal{L}}{\partial \brac{z}{\ell}} = \frac{\partial \mathcal{L}}{\partial \brac{z}{L}} \prod_{k=\ell+1}^{L} \brac{W}{k} \sigma'(\brac{z}{k-1})
    \end{equation}
    but each term in the product is bounded by \(\frac{1}{4}\). taking the norm on both sides gives 

    \begin{equation}
        \frac{\partial \mathcal{L}}{\partial z^{[\ell]}} \le \left( \frac{1}{4} \right)^{L-\ell} \cdot \frac{\partial \mathcal{L}}{\partial z^{[L]}} \cdot \prod_{k=\ell+1}^{L} \parallel W^{[k]} \parallel
    \end{equation}

    computing for \(L=10, \ell = 1\), 

    $$\left(\frac{1}{4}\right)^{9} = \frac{1}{262,144} \approx \mathbf{3.81 \times 10^{-6}}$$

    \item $$\text{ReLU}'(x) = \begin{cases} 1 & \text{if } x > 0 \\ 0 & \text{if } x < 0 \end{cases}$$
